// Pulls lineups, matchups and team info from MyFantasyLeague. // League IDs and API keys: Settings!A2:B3. // Week override (optional): Settings!D2. Leave blank to use MFL's current week. const YEAR = 2026; const BUDGET_MS = 240000; // stop cleanly after 4 minutes (Google kills scripts at 6) var START = Date.now(); const LINEUP_HEADERS = ['Week', 'League', 'Franchise ID', 'Team', 'Player ID', 'Player', 'Position', 'NFL team', 'Status', 'Score', 'Secs left']; const MATCHUP_HEADERS = ['Week', 'League', 'Franchise A ID', 'Team A', 'Franchise B ID', 'Team B']; const TEAM_HEADERS = ['League', 'Franchise ID', 'Team', 'Division ID', 'Division', 'Conference']; function onOpen() { SpreadsheetApp.getUi().createMenu('Roster Wars') .addItem('Pull lineups and matchups now', 'pullLineups') .addItem('Load all past weeks of lineups', 'backfillSeason') .addItem('Refresh every 30 minutes', 'installTrigger') .addItem('Refresh every 10 minutes (game days)', 'installTrigger10') .addToUi(); } function installTrigger() { ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'pullLineups') ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger('pullLineups').timeBased().everyMinutes(30).create(); } function installTrigger10() { ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'pullLineups') ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger('pullLineups').timeBased().everyMinutes(10).create(); } // Live scoring: current points and game seconds remaining per player // (3600 = game not started, 0 = game over). Returns {} if MFL does not answer. function liveMap(id, key, week) { var map = {}; try { var root = mflGet({ TYPE: 'liveScoring', L: id, W: week, DETAILS: 1 }, key); var fid = ''; walk(root, function (el) { var n = el.getName(); if (n === 'franchise') fid = attr(el, 'id'); if (n === 'player' && fid !== '' && attr(el, 'gameSecondsRemaining') !== '') { map[fid + '|' + attr(el, 'id')] = { score: attr(el, 'score'), secs: attr(el, 'gameSecondsRemaining') }; } }); } catch (err) { /* live data is optional; the regular pull still works */ } return map; } function mflGet(params, key) { var qs = Object.keys(params).map(function (k) { return k + '=' + encodeURIComponent(params[k]); }).join('&'); var url = 'https://www.myfantasyleague.com/' + YEAR + '/export?' + qs + '&APIKEY=' + encodeURIComponent(key); var res, code; for (var attempt = 1; attempt <= 2; attempt++) { if (Date.now() - START > BUDGET_MS) throw new Error('out of time budget'); try { res = UrlFetchApp.fetch(url, { muteHttpExceptions: true }); code = res.getResponseCode(); if (code === 200) break; } catch (e) { code = String(e); } if (attempt < 2) Utilities.sleep(3000); } if (code !== 200) { throw new Error('MFL returned ' + code + ' for ' + params.TYPE); } return XmlService.parse(res.getContentText()).getRootElement(); } // Several MFL requests at once. Returns one entry per request: the XML root, // or null when that request failed (after one retry on its own). function mflFetchAll(reqs) { var reqObjs = reqs.map(function (r) { var qs = Object.keys(r.params).map(function (k) { return k + '=' + encodeURIComponent(r.params[k]); }).join('&'); return { url: 'https://www.myfantasyleague.com/' + YEAR + '/export?' + qs + '&APIKEY=' + encodeURIComponent(r.key), muteHttpExceptions: true }; }); var resps = []; try { resps = UrlFetchApp.fetchAll(reqObjs); } catch (e) { resps = []; } return reqs.map(function (r, i) { try { if (resps[i] && resps[i].getResponseCode() === 200) { return XmlService.parse(resps[i].getContentText()).getRootElement(); } } catch (e) { /* fall through to a single retry */ } try { return mflGet(r.params, r.key); } catch (e2) { return null; } }); } function liveFromRoot(root) { var map = {}; if (!root) return map; var fid = ''; walk(root, function (el) { var n = el.getName(); if (n === 'franchise') fid = attr(el, 'id'); if (n === 'player' && fid !== '' && attr(el, 'gameSecondsRemaining') !== '') { map[fid + '|' + attr(el, 'id')] = { score: attr(el, 'score'), secs: attr(el, 'gameSecondsRemaining') }; } }); return map; } function walk(el, fn) { fn(el); el.getChildren().forEach(function (c) { walk(c, fn); }); } function attr(el, name) { var a = el.getAttribute(name); return a ? a.getValue() : ''; } function playerMap() { var sh = SpreadsheetApp.getActive().getSheetByName('Players'); var map = {}; sh.getRange(2, 1, sh.getLastRow() - 1, 4).getValues().forEach(function (r) { if (r[0] !== '') map[String(r[0])] = { name: r[1], pos: r[2], nfl: r[3] }; }); return map; } function sheetFor(name) { var ss = SpreadsheetApp.getActive(); return ss.getSheetByName(name) || ss.insertSheet(name); } function readRows(sh, ncols) { return sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, ncols).getValues() : []; } function writeTable(sh, headers, rows) { var oldLast = sh.getLastRow(); if (oldLast > 1) sh.getRange(2, 1, oldLast - 1, headers.length).clearContent(); sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold'); if (rows.length) sh.getRange(2, 1, rows.length, headers.length).setValues(rows); sh.setFrozenRows(1); } function pullLineups(e) { // never let two runs overlap: a slow run used to pile up behind the next one var lock = LockService.getScriptLock(); if (!lock.tryLock(5000)) return; try { pullLineupsBody_(e); } finally { lock.releaseLock(); } } function pullLineupsBody_(e) { START = Date.now(); var auto = !!(e && e.triggerUid); // true when run by the schedule var ss = SpreadsheetApp.getActive(); var settings = ss.getSheetByName('Settings'); var leagues = settings.getRange('A2:B3').getValues() .filter(function (r) { return r[0] !== '' && r[1] !== ''; }); var override = String(settings.getRange('D2').getValue() || '').trim(); var players = playerMap(); var lineSh = sheetFor('Lineups'); var matchSh = sheetFor('Matchups'); var teamSh = sheetFor('Teams'); var matchRows = readRows(matchSh, MATCHUP_HEADERS.length); var oldTeams = readRows(teamSh, TEAM_HEADERS.length); var teamRows = []; var results = []; // {id, week, rows} for each league that answered var log = []; // Stage 1: everything the 30-minute run needs, fetched in parallel. var jobs = []; leagues.forEach(function (lg) { var id = String(lg[0]), key = lg[1]; var haveTeams = oldTeams.filter(function (r) { return String(r[0]) === id; }).length > 0; var wr = { TYPE: 'weeklyResults', L: id }; var lv = { TYPE: 'liveScoring', L: id, DETAILS: 1 }; if (override !== '') { wr.W = override; lv.W = override; } jobs.push({ id: id, kind: 'wr', req: { params: wr, key: key } }); jobs.push({ id: id, kind: 'live', req: { params: lv, key: key } }); if (!auto || !haveTeams) jobs.push({ id: id, kind: 'league', req: { params: { TYPE: 'league', L: id }, key: key } }); }); var fetched = mflFetchAll(jobs.map(function (j) { return j.req; })); var got = {}; jobs.forEach(function (j, i) { got[j.id + '|' + j.kind] = fetched[i]; }); leagues.forEach(function (lg) { var id = String(lg[0]), key = lg[1]; try { // team names and divisions (scheduled runs reuse the saved Teams tab) var names = {}, divOf = {}, divNames = {}; var leagueRoot = got[id + '|league']; if (leagueRoot) { walk(leagueRoot, function (el) { var n = el.getName(); if (n === 'franchise') { names[attr(el, 'id')] = attr(el, 'name'); divOf[attr(el, 'id')] = attr(el, 'division'); } if (n === 'division') divNames[attr(el, 'id')] = attr(el, 'name'); }); Object.keys(names).forEach(function (f) { teamRows.push([id, f, names[f], divOf[f] || '', divNames[divOf[f]] || '']); }); } else { oldTeams.filter(function (r) { return String(r[0]) === id; }).forEach(function (r) { names[String(r[1])] = r[2]; teamRows.push(r.slice(0, 5)); }); if (!Object.keys(names).length) throw new Error('no team list'); } // lineups (and the current week when no override is set) var root = got[id + '|wr']; if (!root) throw new Error('weeklyResults did not answer'); var week = override !== '' ? override : attr(root, 'week'); var rows = []; var fid = ''; var live = liveFromRoot(got[id + '|live']); walk(root, function (el) { var n = el.getName(); if (n === 'franchise') fid = attr(el, 'id'); if (n === 'player' && attr(el, 'status') !== '' && fid !== '') { var pid = attr(el, 'id'); var p = players[pid] || { name: pid, pos: '', nfl: '' }; var lv = live[fid + '|' + pid]; var score = lv && lv.score !== '' ? lv.score : attr(el, 'score'); rows.push([Number(week), id, fid, names[fid] || fid, pid, p.name, p.pos, p.nfl, attr(el, 'status'), score, lv ? lv.secs : '']); } }); results.push({ id: id, week: Number(week), rows: rows }); // matchups for the whole season var pairs = []; var curWeek = ''; function collect(sched, forcedWeek) { walk(sched, function (el) { var n = el.getName(); if (n === 'weeklySchedule') curWeek = attr(el, 'week'); if (n === 'matchup') { var fr = el.getChildren('franchise'); if (fr.length === 2) { var a = attr(fr[0], 'id'), b = attr(fr[1], 'id'); pairs.push([Number(forcedWeek || curWeek), id, a, names[a] || a, b, names[b] || b]); } } }); } var haveSchedule = matchRows.filter(function (r) { return String(r[1]) === id; }).length; if (!auto || haveSchedule < 100) { // the schedule rarely changes: scheduled runs skip it once loaded collect(mflGet({ TYPE: 'schedule', L: id }, key), ''); if (pairs.length === 0) { for (var w = 1; w <= 18; w++) { collect(mflGet({ TYPE: 'schedule', L: id, W: w }, key), w); } } matchRows = matchRows.filter(function (r) { return String(r[1]) !== id; }) .concat(pairs); } else { pairs = matchRows.filter(function (r) { return String(r[1]) === id; }); } log.push('League ' + id + ' week ' + week + ': ' + rows.length + ' lineup rows, ' + pairs.length + ' season matchups'); } catch (err) { // keep this league's previous team rows so the Conference column stays lined up teamRows = teamRows.filter(function (r) { return String(r[0]) !== id; }) .concat(oldTeams.filter(function (r) { return String(r[0]) === id; }) .map(function (r) { return r.slice(0, 5); })); log.push('League ' + id + ' SKIPPED (' + err.message + '), kept the previous team data'); } }); var fetchSecs = Math.round((Date.now() - START) / 1000); // Lineups: only the current week's rows (always at the bottom) are rewritten, // not the whole season, so a run stays fast as the sheet grows. if (results.length) { var N = LINEUP_HEADERS.length; var minWeek = Math.min.apply(null, results.map(function (r) { return r.week; })); var last = lineSh.getLastRow(); var colA = last > 1 ? lineSh.getRange(2, 1, last - 1, 1).getValues() : []; var startIdx = colA.length; for (var i = 0; i < colA.length; i++) { if (Number(colA[i][0]) >= minWeek) { startIdx = i; break; } } var startRow = 2 + startIdx; var oldCount = colA.length - startIdx; var tail = oldCount > 0 ? lineSh.getRange(startRow, 1, oldCount, N).getValues() : []; results.forEach(function (r) { tail = tail.filter(function (row) { return !(String(row[0]) === String(r.week) && String(row[1]) === r.id); }).concat(r.rows); }); if (oldCount > 0) lineSh.getRange(startRow, 1, oldCount, N).clearContent(); var needRows = startRow + tail.length - 1; if (needRows > lineSh.getMaxRows()) lineSh.insertRowsAfter(lineSh.getMaxRows(), needRows - lineSh.getMaxRows()); if (tail.length) lineSh.getRange(startRow, 1, tail.length, N).setValues(tail); lineSh.getRange(1, 1, 1, N).setValues([LINEUP_HEADERS]).setFontWeight('bold'); lineSh.setFrozenRows(1); } writeTable(matchSh, MATCHUP_HEADERS, matchRows); // columns F onward (Conference and the display name/key formulas) are left alone writeTable(teamSh, TEAM_HEADERS.slice(0, 5), teamRows); var totalSecs = Math.round((Date.now() - START) / 1000); lineSh.getRange('L1').setValue('Last pull: ' + new Date() + ' | fetch ' + fetchSecs + 's, total ' + totalSecs + 's | ' + log.join('; ')); } // One-time: loads lineups and scores for every week before the current one. function backfillSeason() { var ss = SpreadsheetApp.getActive(); var settings = ss.getSheetByName('Settings'); var leagues = settings.getRange('A2:B3').getValues() .filter(function (r) { return r[0] !== '' && r[1] !== ''; }); var players = playerMap(); var lineSh = sheetFor('Lineups'); var lineRows = readRows(lineSh, LINEUP_HEADERS.length); var log = []; leagues.forEach(function (lg) { var id = String(lg[0]), key = lg[1]; var names = {}; walk(mflGet({ TYPE: 'league', L: id }, key), function (el) { if (el.getName() === 'franchise') names[attr(el, 'id')] = attr(el, 'name'); }); var cur = Number(attr(mflGet({ TYPE: 'weeklyResults', L: id }, key), 'week')); for (var week = 1; week <= cur; week++) { var root = mflGet({ TYPE: 'weeklyResults', L: id, W: week }, key); var rows = []; var fid = ''; walk(root, function (el) { var n = el.getName(); if (n === 'franchise') fid = attr(el, 'id'); if (n === 'player' && attr(el, 'status') !== '' && fid !== '') { var pid = attr(el, 'id'); var p = players[pid] || { name: pid, pos: '', nfl: '' }; rows.push([week, id, fid, names[fid] || fid, pid, p.name, p.pos, p.nfl, attr(el, 'status'), attr(el, 'score'), '']); } }); lineRows = lineRows.filter(function (r) { return !(String(r[0]) === String(week) && String(r[1]) === id); }).concat(rows); log.push(id + ' wk' + week + ':' + rows.length); } }); writeTable(lineSh, LINEUP_HEADERS, lineRows); lineSh.getRange('L2').setValue('Backfill: ' + new Date() + ' | ' + log.join(', ')); }
